`timescale 1ns/1ps

module Elevador_tb;

    // Entradas
    reg CLOCK_50;
    reg [9:0] SW;
    reg [3:0] KEY;

    // Saídas
    wire [6:0] HEX0, HEX1, HEX2, HEX3;
    wire [9:0] LEDR;

    // Instancia o DUT (Device Under Test)
    Elevador dut (
        .CLOCK_50(CLOCK_50),
        .SW(SW),
        .KEY(KEY),
        .HEX0(HEX0),
        .HEX1(HEX1),
        .HEX2(HEX2),
        .HEX3(HEX3),
        .LEDR(LEDR)
    );

    // =========================================================
    //  Geração do clock de 50 MHz (período de 20 ns)
    // =========================================================
    initial begin
        CLOCK_50 = 0;
        forever #10 CLOCK_50 = ~CLOCK_50; // 50 MHz
    end

    // =========================================================
    //  Sequência de testes
    // =========================================================
    initial begin
        // Inicializa sinais
        SW = 10'b0;
        KEY = 4'b1111;  // Todas as teclas soltas
        #100;

        // Aplica RESET (KEY1 = 0 → reset ativo)
        KEY[1] = 0;
        #200;
        KEY[1] = 1; // Libera reset
        #200;

        // Simula pessoa solicitando andar 3
        SW[7] = 1; // Andar 3
        #100000000; // Espera alguns ciclos (simula o movimento do elevador)
        SW[7] = 0;

        // Solicita andar 5
        SW[9] = 1;
        #100000000;
        SW[9] = 0;

        // Ativa emergência via SW3
        SW[3] = 1;
        #50000000;
        SW[3] = 0;

        // Solicita andar 1 (descendo)
        SW[5] = 1;
        #100000000;
        SW[5] = 0;

        // Fim da simulação
        #100000000;
        $stop;
    end

    // =========================================================
    //  Monitoramento (opcional)
    // =========================================================
    initial begin
        $display("Tempo\tAndar\tReq\tDir\tEmergencia");
        $monitor("%0t\t%b\t%b\t%b\t%b",
                 $time, dut.andar, dut.req_andar, dut.direcao, dut.emergencia);
    end

endmodule
