void bubbleSort(int n, int *vet, int *comparacoes) {
    int i, j, temp;
    *comparacoes = 0;

    for (i = 0; i < n - 1; i++) {
        for (j = 0; j < n - i - 1; j++) {
            (*comparacoes)++; // comparação
            if (vet[j] > vet[j + 1]) {
                temp = vet[j];
                vet[j] = vet[j + 1];
                vet[j + 1] = temp;
            }
        }
    }
}

// Insertion Sort
void insertionSort(int n, int *vet, int *comparacoes) {
    int i, j, chave;
    *comparacoes = 0;

    for (i = 1; i < n; i++) {
        chave = vet[i];
        j = i - 1;
        while (j >= 0) {
            (*comparacoes)++; // comparação
            if (vet[j] > chave) {
                vet[j + 1] = vet[j];
                j--;
            } else {
                break;
            }
        }
        vet[j + 1] = chave;
    }
}
